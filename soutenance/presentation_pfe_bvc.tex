\begin{frame}{1. Contexte et problematique \duration{2 min}}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.34\textwidth}
      \metric{Seances analysees}{247}{Annee 2025}
      \metric{Instruments}{80}{Titres presents dans le dataset}
      \metric{Indice de reference}{MASI}{Lecture macro du marche}
    \end{column}
    \begin{column}{0.62\textwidth}
      \begin{block}{Enjeux de la surveillance}
        Preserver l'integrite du marche, identifier les ruptures de liquidite, detecter les volumes ou rendements atypiques et produire des alertes actionnables.
      \end{block}
      \begin{itemize}
        \item Les regles statiques seules s'adaptent mal aux regimes de marche.
        \item Les donnees de microstructure apportent un signal fin : transactions, volumes, OIR, OAR, VWAP.
      \end{itemize}
      \begin{block}{Problematique}
        Comment detecter automatiquement les comportements atypiques a partir des donnees disponibles ?
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{3. Indicateurs calcules \duration{1 min 20}}

    \bottomrule

  \vspace{0.8em}
  \[
    OIR_t=\frac{V^{buy}_t-V^{sell}_t}{V^{buy}_t+V^{sell}_t}
    \qquad
    VWAP_t=\frac{\sum_i P_iV_i}{\sum_i V_i}
  \]
  \[
    VaR_{\alpha}=-Quantile_{1-\alpha}(r_t)
  \]
\end{frame}

% Slide 7
\begin{frame}{3. Score de stress de marche \duration{1 min 20}}
  \begin{block}{Trois composantes}
    Volatilite du MASI, deterioration de la breadth, desequilibre du flux d'ordres.
  \end{block}
  \[
    S_{stress}=\frac{S_{vol}+S_{breadth}+S_{OIR}}{3}
  \]
  \[
    S_{vol}=rang(\sigma_{MASI,20j})\times100,\quad
    S_{breadth}=100-Breadth_t,\quad
    S_{OIR}=rang(|OIR|_t)\times100
  \]
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \metric{Exemple de lecture}{72 / 100}{Marche sous tension elevee}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Interet}
        Un seul score synthetise volatilite, largeur de marche et pression intraday.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% Slide 8
\begin{frame}{3. Detection statistique \duration{1 min 20}}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.46\textwidth}
      \begin{block}{6 familles de methodes}
        Z-score, z-score rolling, IQR, percentiles, bandes de Bollinger, CUSUM.
      \end{block}
      \begin{itemize}
        \item Chaque methode vote si l'observation est atypique.
        \item Le consensus combine nombre de votes et intensite du signal.
        \item Severite finale : Normal, Faible, Modere, Critique.
      \end{itemize}
    \end{column}

% Slide 9
\begin{frame}{4. Modeles ML non supervises \duration{1 min 30}}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Isolation Forest}
        Partitionnements aleatoires : une anomalie est isolee avec moins de coupes.
        Contamination fixee a \textbf{0.05}. Score normalise dans $[0,1]$.
      \end{block}
      \begin{block}{Autoencoder}
        Architecture dense \textbf{7--8--7}. Une forte erreur de reconstruction indique un profil inhabituel.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Score ML combine}
        \[
          Score_{ML}=0.6\,Score_{IF}+0.4\,Score_{AE}
        \]
      \end{block}
      \repofig[width=\textwidth,height=0.38\textheight,keepaspectratio]{ml_02_if_scores_marche.png}
    \end{column}
  \end{columns}
\end{frame}

% Slide 10
\begin{frame}{4. Pourquoi du ML non supervise ? \duration{1 min 30}}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.46\textwidth}
      \begin{itemize}
        \item Pas de labels metier exhaustifs pour entrainer un modele supervise.
        \item Objectif : completer les regles statistiques, pas les remplacer.
        \item Les seuils ML structurent la priorisation : 40, 60, 80.
      \end{itemize}
      \begin{block}{Severite ML}
        Faible $\geq 40$ -- Modere $\geq 60$ -- Critique $\geq 80$
      \end{block}
    \end{column}
    \begin{column}{0.50\textwidth}
      \repofig[width=\textwidth,height=0.61\textheight,keepaspectratio]{ml_04_comparaison_stat_ml.png}
    \end{column}
  \end{columns}
\end{frame}
